{/* SNIPPET: copy-ai-prompt — Card com uma frase curta e um botão verde "vivo" ao lado que copia o prompt de IA.
    O texto do prompt não aparece; só é exibido (e selecionado) se o navegador bloquear a cópia.
    Animação e cores do botão ficam no style.css (.judit-ai-prompt / .judit-ai-prompt-btn). */}

export const CopyAiPrompt = ({
  prompt,
  title,
  description,
  buttonLabel = "Copiar prompt",
  copiedLabel = "Copiado!",
  fallbackLabel = "Selecione e copie",
}) => {
  const [state, setState] = useState("idle");
  const promptRef = useRef(null);

  const selectPrompt = () => {
    const el = promptRef.current;
    if (!el || typeof window === "undefined" || !window.getSelection) return;
    const range = document.createRange();
    range.selectNodeContents(el);
    const sel = window.getSelection();
    sel.removeAllRanges();
    sel.addRange(range);
  };

  useEffect(() => {
    if (state === "fallback") selectPrompt();
    if (state !== "copied") return;
    const t = setTimeout(() => setState("idle"), 2000);
    return () => clearTimeout(t);
  }, [state]);

  const handleCopy = async () => {
    try {
      if (!navigator.clipboard || !navigator.clipboard.writeText) throw new Error("clipboard indisponível");
      await navigator.clipboard.writeText(prompt);
      setState("copied");
    } catch (e) {
      setState("fallback");
    }
  };

  const label = state === "copied" ? copiedLabel : state === "fallback" ? fallbackLabel : buttonLabel;
  const icon = state === "copied" ? "✓" : "✨";

  return (
    <div className="judit-ai-prompt" style={{ borderRadius: "12px", padding: "16px 20px", margin: "8px 0 24px" }}>
      <div style={{ display: "flex", flexWrap: "wrap", alignItems: "center", justifyContent: "space-between", gap: "12px 20px" }}>
        <div style={{ flex: "1 1 320px", minWidth: 0 }}>
          <div style={{ display: "flex", alignItems: "center", gap: "8px", fontWeight: 600, fontSize: "16px" }}>
            <span aria-hidden="true">🤖</span>
            <span>{title}</span>
          </div>
          {description && (
            <p style={{ margin: "4px 0 0", fontSize: "14px", opacity: 0.85 }}>{description}</p>
          )}
        </div>
        <button type="button" className="judit-ai-prompt-btn" onClick={handleCopy}>
          <span aria-hidden="true">{icon}</span>
          <span aria-live="polite">{label}</span>
        </button>
      </div>
      {state === "fallback" && (
        <code
          ref={promptRef}
          style={{
            display: "block",
            marginTop: "12px",
            padding: "10px 12px",
            borderRadius: "8px",
            fontSize: "13px",
            lineHeight: 1.5,
            whiteSpace: "normal",
            wordBreak: "break-word",
            userSelect: "all",
          }}
        >
          {prompt}
        </code>
      )}
    </div>
  );
};

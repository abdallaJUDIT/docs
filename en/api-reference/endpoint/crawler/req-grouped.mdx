openapi: "openapi-1 POST /requests"

---

    <Expandable title="properties">

        <ParamField body="search_type" type="string" required>

                                </ParamField>
                                <ParamField body="not_equal" name="not_equal" type="boolean" required={false}>
                                    Determines whether the filter matches lawsuits that contain or do not contain the provided codes.
                                </ParamField>

                        <ParamField body="subject_codes" name="subject_codes" type="object" required={false}>

                                <ParamField body="keys" name="keys" type="array" required={false}>
                                    Filters lawsuits with (or without) these codes.

                                <ParamField body="not_equal" name="not_equal" type="boolean" required={false}>
                                    Determines whether the filter matches lawsuits that contain or do not contain the provided codes.
                                </ParamField>
                            </Expandable>
                        </ParamField>

                        <ParamField body="classification_codes" name="classification_codes" type="object" required={false}>
                            <Expandable title="properties">
                                <ParamField body="keys" name="keys" type="array" required={false}>
                                    Filters lawsuits with (or without) these codes.
                                </ParamField>
                                <ParamField body="not_equal" name="not_equal" type="boolean" required={false}>
                                    Determines whether the filter matches lawsuits that contain or do not contain the provided codes.
                                </ParamField>
                            </Expandable>
                        </ParamField>

                        <ParamField body="last_step_date_gte" name="last_step_date_gte" type="date" required={false}>
                            Returns searches whose last step date is greater than or equal to the value provided.
                        </ParamField>

                        <ParamField body="last_step_date_lte" name="last_step_date_lte" type="date" required={false}>
                            Returns searches whose last step date is less than or equal to the value provided.
                        </ParamField>

                        <ParamField body="party_names" name="party_names" type="array" required={false}>
                            List of names (array of strings) that restricts the search to lawsuits containing any of them in one of the parties.
                        </ParamField>

                        <ParamField body="party_documents" name="party_documents" type="array" required={false}>
                            List of documents (array of strings) that restricts the search to lawsuits containing any of them in one of the parties.

                    </Expandable>

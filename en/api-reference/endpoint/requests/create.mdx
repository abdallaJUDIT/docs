---
title: "Create a request"
openapi: "openapi-1 POST /requests"
api: "POST https://requests.prod.judit.io/requests"
description: "This endpoint creates a lawsuit request by document (CPF, CNPJ, OAB) or CNJ code."
---

<ParamField body="search" type="object" required={false}>
    <Expandable title="properties">

        <ParamField body="search_type" type="string" required>
            Defines the type of entity you want to search: `cpf`, `cnpj`, `oab`, `lawsuit_cnj`, or `lawsuit_id`.
        </ParamField>

        <ParamField body="search_key" type="string" required>
            Lawsuit number (CNJ code), CPF, CNPJ, or OAB.
        </ParamField>

        <ParamField body="response_type" type="string" required={false}>
            Defines the type of the search response data: `lawsuit`, `parties`, `attachments`, `step`, or `lawsuits`.
        </ParamField>

        <ParamField body="cache_ttl_in_days" type="integer" required={false}>
            Defines how many days the search result can still be considered a valid cache.
        </ParamField>

        <ParamField body="should_search_branches" name="should_search_branches" type="boolean" required={false}>
            When `true`, a separate request will be created for each branch (subsidiary) linked to the provided CNPJ.
            All generated requests are associated with the original request through the `origin_id` field.
        </ParamField>

        <ParamField body="search_params" type="object" required={false}>
            <Expandable title="properties">
                <ParamField body="lawsuit_instance" name="lawsuit_instance" type="integer" required={false}>
                    Defines the instance in which you want to search the lawsuit.
                </ParamField>

                <ParamField body="masked_response" name="masked_response" type="boolean" required={false}>
                    Defines whether the response will come minified. This parameter is used exclusively for simple or complete document queries.
                </ParamField>

                <ParamField body="filter" type="object" required={false}>
                    <Expandable title="properties">
                        <ParamField body="side" name="side" type="string" required={false}>
                            Defines the types of participants in the lawsuit: `Passive` or `Active`.
                        </ParamField>

                        <ParamField body="distribution_date_gte" name="distribution_date_gte" type="date" required={false}>
                            Returns searches with a distribution date greater than or equal to the one provided.
                        </ParamField>

                        <ParamField body="amount_gte" name="amount_gte" type="number" required={false}>
                            Returns searches whose case value is greater than the value provided.
                        </ParamField>

                        <ParamField body="amount_lte" name="amount_lte" type="number" required={false}>
                            Returns searches whose case value is less than the value provided.
                        </ParamField>

                        <ParamField body="tribunals" name="tribunals" type="object" required={false}>
                            <Expandable title="properties">
                                <ParamField body="keys" name="keys" type="array" required={false}>
                                    Filters lawsuits with (or without) these codes.
                                </ParamField>
                                <ParamField body="not_equal" name="not_equal" type="boolean" required={false}>
                                    Determines whether the filter matches lawsuits that contain or do not contain the provided codes.
                                </ParamField>
                            </Expandable>
                        </ParamField>

                        <ParamField body="subject_codes" name="subject_codes" type="object" required={false}>
                            <Expandable title="properties">
                                <ParamField body="keys" name="keys" type="array" required={false}>
                                    Filters lawsuits with (or without) these codes.
                                </ParamField>
                                <ParamField body="not_equal" name="not_equal" type="boolean" required={false}>
                                    Determines whether the filter matches lawsuits that contain or do not contain the provided codes.
                                </ParamField>
                            </Expandable>
                        </ParamField>

                        <ParamField body="classification_codes" name="classification_codes" type="object" required={false}>
                            <Expandable title="properties">
                                <ParamField body="keys" name="keys" type="array" required={false}>
                                    Filters lawsuits with (or without) these codes.
                                </ParamField>
                                <ParamField body="not_equal" name="not_equal" type="boolean" required={false}>
                                    Determines whether the filter matches lawsuits that contain or do not contain the provided codes.
                                </ParamField>
                            </Expandable>
                        </ParamField>
                    </Expandable>
                </ParamField>

                <ParamField body="credential" type="object" required={false}>
                    <Expandable title="properties">
                        <ParamField body="customer_key" name="customer_key" type="string" required={false}>
                            Lets you pass the client's key to access the correct credentials vault. If not provided, the API will try to find a credential registered with an empty `customer_key`.
                        </ParamField>
                    </Expandable>
                </ParamField>
            </Expandable>
        </ParamField>
    </Expandable>

</ParamField>

---
title: "List responses"
openapi: "GET /responses"
api: "GET https://requests.prod.judit.io/responses"
description: "This endpoint queries the responses of requests by parameters, in a paginated way."
---

<ParamField query="page" type="integer" required={false}>
    Defines which page of results you want to fetch.
</ParamField>

<ParamField query="page_size" type="integer" required={false}>
    Defines the maximum number of results you want to receive per page.
</ParamField>

<ParamField query="orderField" type="string" required={false}>
    Name of the field to sort the responses by.
</ParamField>

<ParamField query="orderDirection" type="string" required={false}>
    Sort direction: 'asc' or 'desc'.
</ParamField>

<ParamField query="response_type" type="string" required={false}>
    Returns responses of the specified type: `lawsuit`, `parties`, `attachments`, `steps`, or `lawsuits`.
</ParamField>

<ParamField query="request_id" type="string" required={false}>
    Returns responses related to a request by its id.
</ParamField>

<ParamField query="request_status" type="string" required={false}>
    Returns responses whose request status is one of `created`, `pending`, `started`, `cancelling`, `cancelled`, or `completed`. You may also pass more than one status as an array (e.g., `['started', 'completed']`).
</ParamField>

<ParamField query="created_at_gte" type="date" required={false}>
    Returns responses whose creation date is greater than or equal to the value provided.
</ParamField>

<ParamField query="created_at_lte" type="date" required={false}>
    Returns responses whose creation date is less than or equal to the value provided.
</ParamField>

<ParamField query="request_created_at_gte" type="date" required={false}>
    Returns responses whose request creation date is greater than or equal to the value provided.
</ParamField>

<ParamField query="request_created_at_lte" type="date" required={false}>
    Returns responses whose request creation date is less than or equal to the value provided.
</ParamField>
